== definicao da view adm360_saldos ==
   SELECT m.tenant_id,
      m.produto_id,
      p.nome AS produto,
      m.lote_id,
      l.codigo AS lote,
      l.validade,
      m.local_id,
      o.nome AS local,
      m.condicao,
      (sum(m.quantidade))::numeric(18,4) AS fisico,
      (COALESCE(( SELECT sum(r.quantidade) AS sum
             FROM plantaopro.adm360_reservas r
            WHERE ((r.tenant_id = m.tenant_id) AND (r.produto_id = m.produto_id) AND (r.lote_id = m.lote_id) AND (r.local_id = m.local_id) AND ((r.situacao)::text = 'ATIVA'::text))), (0)::numeric))::numeric(18,4) AS reservado,
      (
          CASE
              WHEN (((m.condicao)::text = 'LIBERADO'::text) AND ((l.validade IS NULL) OR (l.validade >= CURRENT_DATE))) THEN (sum(m.quantidade) - COALESCE(( SELECT sum(r.quantidade) AS sum
                 FROM plantaopro.adm360_reservas r
                WHERE ((r.tenant_id = m.tenant_id) AND (r.produto_id = m.produto_id) AND (r.lote_id = m.lote_id) AND (r.local_id = m.local_id) AND ((r.situacao)::text = 'ATIVA'::text))), (0)::numeric))
              ELSE (0)::numeric
          END)::numeric(18,4) AS disponivel
     FROM (((plantaopro.adm360_movimentos m
       JOIN plantaopro.adm360_produtos p ON (((p.id = m.produto_id) AND (p.tenant_id = m.tenant_id))))
       JOIN plantaopro.adm360_lotes l ON (((l.id = m.lote_id) AND (l.tenant_id = m.tenant_id))))
       JOIN plantaopro.adm360_locais o ON (((o.id = m.local_id) AND (o.tenant_id = m.tenant_id))))
    GROUP BY m.tenant_id, m.produto_id, p.nome, m.lote_id, l.codigo, l.validade, m.local_id, o.nome, m.condicao;
